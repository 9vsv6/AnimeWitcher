            '          } else if (part.complete) {\n'
            '            await saveRecord(\n'
            '              TaskRecord(part.task, TaskStatus.complete, 1, part.size),\n'
            '            );\n'
            '          }\n'
            '        }\n'
            '\n'
            '        if (repaired) {\n'
            '          await _persist(session);\n'
            '          final parentRecord = await recordForId(task.taskId);\n'
            '          if (parentRecord != null &&\n'
            '              parentRecord.status != TaskStatus.complete) {\n'
            '            await saveRecord(\n'
            '              TaskRecord(\n'
            '                task,\n'
            '                parentRecord.status,\n'
            '                session.progress,\n'
            '                session.size,\n'
            '              ),\n'
            '            );\n'
            '          }\n'
            '        }\n'
            '        return true;\n'
            '      } catch (_) {\n'
            '        // Try the .tmp checkpoint when the primary manifest was torn/corrupt.\n'
            '      }\n'
            '    }\n'
            '    return false;\n'
            '  }\n'
            '\n'
            '  void _register(_ParallelSession session) {\n'
            '    _sessions[session.task.taskId] = session;\n'
            '    for (final part in session.parts) {\n'
            '      _children[part.task.taskId] = session;\n'
            '    }\n'
            '  }\n'
            '\n'
            '  Future<void> _restoreNativeOwnership(_ParallelSession session) async {\n'
            '    final live = await livePartIds?.call() ?? <String>{};\n'
            '    for (final part in session.parts) {\n'
            '      part.launched = !part.complete && live.contains(part.task.taskId);\n'
            '      if (part.launched) {\n'
            '        _activeConnectionIds.add(part.task.taskId);\n'
            '      } else {\n'
            '        _activeConnectionIds.remove(part.task.taskId);\n'
            '      }\n'
            '    }\n'
            '  }\n'
            '\n'
            '  /// Imports legacy plugin checkpoints without cancelling/deleting their files.\n'
            '  /// Completed parts keep their filenames; the remaining native resume blobs\n'
            '  /// stay associated with the same child taskIds.\n'
            '  Future<void> importLegacy(\n'
            '    ParallelDownloadTask task,\n'
            '    String resumeData,\n'
            '  ) async {\n'
            '    if (_disposed) return;\n'
            '    if (await restore(task)) return;\n'
            '    final chunks = jsonDecode(resumeData) as List;\n'
            '    if (chunks.isEmpty) throw const FormatException(\'Empty chunk checkpoint\');\n'
            '    final parts = chunks.map((raw) {\n'
            '      final chunk = Map<String, dynamic>.from(raw as Map);\n'
            '      final child = Task.createFromJson(\n'
            '        Map<String, dynamic>.from(chunk[\'task\'] as Map),\n'
            '      ) as DownloadTask;\n'
            '      return _DownloadPart(\n'
            '        child.copyWith(\n'
            '          group: kPersistentDownloadChunkGroup,\n'
            '          retries: kDownloadPartRetries,\n'
            '        ),\n'
            '        (chunk[\'fromByte\'] as num).toInt(),\n'
            '        (chunk[\'toByte\'] as num).toInt(),\n'
            '        progress: (chunk[\'progress\'] as num? ?? 0).toDouble(),\n'
            '        complete: chunk[\'status\'] == TaskStatus.complete.index,\n'
            '      );\n'
            '    }).toList();\n'
            '    final session = _ParallelSession(task, await _manifest(task), parts);\n'
            '    await _persist(session);\n'
            '    _register(session);\n'
            '  }\n'
            '\n'
            '  Future<bool> start(ParallelDownloadTask task, int totalBytes) async {\n'
            '    if (_disposed) return false;\n'
            '    final restored = await restore(task);\n'
            '    if (!restored) {\n'
            '      if (totalBytes <= 0) return false;\n'
            '      final requestedConnections = task.chunks\n'
            '          .clamp(kDownloadPartsMin, kDownloadPartsMax)\n'
            '          .toInt();\n'
            '      final count = selectDownloadWorkUnitCount(\n'
            '        connections: requestedConnections,\n'
            '        totalBytes: totalBytes,\n'
            '      );\n'
            '      final parts = <_DownloadPart>[];\n'
            '      for (var index = 0; index < count; index++) {\n'
            '        final from = totalBytes * index ~/ count;\n'
            '        final to = totalBytes * (index + 1) ~/ count - 1;\n'
            '        final headers = Map<String, String>.from(task.headers)\n'
            '          ..removeWhere((key, _) => key.toLowerCase() == \'range\');\n'
            '        headers[\'Range\'] = \'bytes=$from-$to\';\n'
            '        headers[\'Accept-Encoding\'] = \'identity\';\n'
            '        parts.add(\n'
            '          _DownloadPart(\n'
            '            DownloadTask(\n'
            '              taskId: \'${task.taskId}.part.$index\',\n'
            '              url: task.url,\n'
            '              filename: \'$index.part\',\n'
            '              directory: p.join(task.directory, \'${task.filename}.parts\'),\n'
            '              baseDirectory: task.baseDirectory,\n'
            '              headers: headers,\n'
            '              updates: Updates.statusAndProgress,\n'
            '              retries: kDownloadPartRetries,\n'
            '              allowPause: true,\n'
            '              group: kPersistentDownloadChunkGroup,\n'
            '              metaData: jsonEncode({\'parentTaskId\': task.taskId}),\n'
            '            ),\n'
            '            from,\n'
            '            to,\n'
            '          ),\n'
            '        );\n'
            '      }\n'
            '      final session = _ParallelSession(task, await _manifest(task), parts);\n'
            '      await _persist(session);\n'
            '      _register(session);\n'
            '    }\n'
            '\n'
            '    final session = _sessions[task.taskId]!;\n'
            '    return session.serialize(() async {\n'
            '      if (_disposed) return false;\n'
            '      if (session.active) return true;\n'
            '      session.cancelAggregateProgress();\n'
            '      session.generation++;\n'
            '      session.active = true;\n'
            '      session.resetRamp();\n'
            '      try {\n'
            '        await _status(session, TaskStatus.enqueued);\n'
            '\n'
            '        // Crash window: assembly may already have atomically renamed the final\n'
            '        // file before the parent complete record/cleanup was persisted. Adopt\n'
            '        // that exact-size target instead of assembling or downloading again.\n'
            '        if (await _adoptCompletedTarget(session)) return true;\n'
            '\n'
            '        await _restoreNativeOwnership(session);\n'
            '        for (final part in session.parts) {\n'
            '          part.speed = 0;\n'
            '          part.recoveryAttempts = 0;\n'
            '\n'
            '          // iOS can leave a completed range reported as live/running if the\n'
            '          // final native callback is lost. Exact on-disk bytes are stronger\n'
            '          // evidence than that stale status. Settle the worker first, verify the\n'
            '          // bytes again, then adopt the range without downloading it twice.\n'
            '          if (part.launched) {\n'
            '            if (await _adoptExactSizePart(\n'
            '              session,\n'
            '              part,\n'
            '              settleNativeOwner: true,\n'
            '            )) {\n'
            '              continue;\n'
            '            }\n'
            '            _armTailStallWatch(session, part);\n'
            '            continue;\n'
            '          }\n'
            '\n'
            '          final saved = await canonicalizePartialDownloadFile(\n'
            '            destinationPath: await part.task.filePath(),\n'
            '          );\n'
            '          final file = saved?.file ?? File(await part.task.filePath());\n'
            '          if (await file.exists() && await file.length() == part.size) {\n'
            '            part.complete = true;\n'
            '            part.progress = 1;\n'
            '            part.credibleProgress = 1;\n'
            '            _activeConnectionIds.remove(part.task.taskId);\n'
            '            part.launched = false;\n'
            '            await saveRecord(\n'
            '              TaskRecord(part.task, TaskStatus.complete, 1, part.size),\n'
            '            );\n'
            '            continue;\n'
            '          }\n'
            '          if (saved != null && saved.bytes > 0 && saved.bytes < part.size) {\n'
            '            final diskProgress = saved.bytes / part.size;\n'
            '            if (diskProgress > part.credibleProgress) {\n'
            '              part.credibleProgress = diskProgress;\n'
            '            }\n'
            '          }\n'
            '          if (part.complete) {\n'
            '            await _pause(session);\n'
            '            return false;\n'
            '          }\n'
            '        }\n'
            '\n'
            '        final pending = session.parts.where((part) => !part.complete).length;\n'
            '        if (pending == 0) {\n'
            '          await _assemble(session);\n'
            '          return true;\n'
            '        }\n'
            '\n'
            '        final configured = task.chunks\n'
            '            .clamp(kDownloadPartsMin, kDownloadPartsMax)\n'
            '            .toInt();\n'
            '        final requested = pending < configured ? pending : configured;\n'
            '        session.connectionCeiling = _connectionGovernor.connectionCeilingFor(\n'
            '          session.task.url,\n'
            '          requested: requested,\n'
            '        );\n'
            '        final slowStartTarget = pending < session.connectionCeiling\n'
            '            ? pending\n'
            '            : session.connectionCeiling;\n'
            '        session.rampBatches = downloadConnectionRampBatches(slowStartTarget);\n'
            '\n'
            '        if (!await _pumpSession(session)) {\n'
            '          throw StateError(\'Could not start initial download connection\');\n'
            '        }\n'
            '        _scheduleDiskProgressPoll(session);\n'
            '        await _persist(session);\n'
            '        _schedulePumpAll();\n'
            '        return true;\n'
            '      } catch (_) {\n'
            '        if (session.active && _activeConnectionsForSession(session) > 0) {\n'
            '          _scheduleDiskProgressPoll(session);\n'
            '          _scheduleCoordinatorRecovery(session);\n'
            '          try {\n'
            '            await _status(session, TaskStatus.running);\n'
            '          } catch (_) {}\n'
            '          return true;\n'
            '        }\n'
            '        await _pause(session);\n'
            '        return false;\n'
            '      }\n'
            '    });\n'
            '  }\n'
            '\n'
            '  int _activeConnectionsForSession(_ParallelSession session) => session.parts\n'
            '      .where(\n'
            '        (part) =>\n'
            '            part.launched &&\n'
            '            !part.complete &&\n'
            '            _activeConnectionIds.contains(part.task.taskId),\n'
            '      )\n'
            '      .length;\n'
            '\n'
            '  Iterable<_DownloadPart> _launchableParts(_ParallelSession session) =>\n'
            '      session.parts.where(\n'
            '        (part) =>\n'
            '            !part.complete && !part.launched && part.recoveryTimer == null,\n'
            '      );\n'
            '\n'
            '  int _launchablePartCount(_ParallelSession session) =>\n'
            '      _launchableParts(session).length;\n'
            '\n'
            '  Future<bool> _pumpSession(_ParallelSession session) async {\n'
            '    if (_disposed || !session.active || session.deleted) return true;\n'
            '\n'
            '    while (!_disposed && session.active && !session.deleted) {\n'
            '      if (session.currentBatchRemaining == 0) {\n'
            '        if (session.currentBatchPendingIds.isNotEmpty) return true;\n'
            '        if (session.rampBatchIndex < session.rampBatches.length) {\n'
            '          session.currentBatchRemaining =\n'
            '              session.rampBatches[session.rampBatchIndex++];\n'
            '        } else {\n'
            '          // Slow start reached its safe ceiling. Keep the pipeline full by\n'
            '          // replacing completed or backing-off ranges one-for-one, never\n'
            '          // exceeding that cap.\n'
            '          session.slowStartComplete = true;\n'
            '          final sessionAvailable =\n'
            '              session.connectionCeiling - _activeConnectionsForSession(session);\n'
            '          if (sessionAvailable <= 0) return true;\n'
            '          final remaining = _launchablePartCount(session);\n'
            '          if (remaining <= 0) return true;\n'
            '          session.currentBatchRemaining = remaining < sessionAvailable\n'
            '              ? remaining\n'
            '              : sessionAvailable;\n'
            '        }\n'
            '      }\n'
            '\n'
            '      final globalAvailable = _connectionBudget - _activeConnectionIds.length;\n'
            '      final sessionAvailable =\n'
            '          session.connectionCeiling - _activeConnectionsForSession(session);\n'
            '      final available = globalAvailable < sessionAvailable\n'
            '          ? globalAvailable\n'
            '          : sessionAvailable;\n'
            '      if (available <= 0) return true;\n'
            '\n'
            '      final launchCount = session.currentBatchRemaining < available\n'
            '          ? session.currentBatchRemaining\n'
            '          : available;\n'
            '      final parts = _launchableParts(session)\n'
            '          .take(launchCount)\n'
            '          .toList(growable: false);\n'
            '\n'
            '      // Every unlaunched range may currently be inside its recovery backoff.\n'
            '      // Preserve the batch count and sleep until one timer makes work eligible;\n'
            '      // otherwise a tight microtask pump can spin at 0 B/s.\n'
            '      if (parts.isEmpty) return true;\n'
            '\n'
            '      for (final part in parts) {\n'
            '        if (_disposed) return true;\n'
            '        final record = await recordForId(part.task.taskId);\n'
            '        final progress = record?.progress ?? 0;\n'
            '        if (progress > part.progress && progress <= 1) {\n'
            '          part.progress = progress;\n'
            '        }\n'
            '        // A child DB checkpoint at 0.999 is the same native sentinel as the\n'
            '        // callback. Preserve it as a resume/recovery hint, but never turn it\n'
            '        // into credited bytes. Ordinary (< 0.999) checkpoints remain useful.\n'
            '        if (progress > part.credibleProgress &&\n'
            '            progress < kParallelNativeCompletionSentinel) {\n'
            '          part.credibleProgress = progress;\n'
            '        }\n'
            '\n'
            '        // Reserve before enqueueing to close the enqueue->running race. This\n'
            '        // also keeps 5 episodes x 16 parts from becoming 80 native requests.\n'
            '        part.launched = true;\n'
            '        _activeConnectionIds.add(part.task.taskId);\n'
            '        session.currentBatchPendingIds.add(part.task.taskId);\n'
            '        session.currentBatchRemaining--;\n'
            '\n'
            '        void rollbackUnownedReservation() {\n'
            '          // Reservation happens before startPart to close the enqueue/running\n'
            '          // race. If native never accepts the child, put that slot back into\n'
            '          // the same slow-start batch. Otherwise repeated transient enqueue\n'
            '          // failures consume the batch counter and can strand the episode with\n'
            '          // no launchable work even though its immutable Range still exists.\n'
            '          if (session.currentBatchPendingIds.remove(part.task.taskId)) {\n'
            '            session.currentBatchRemaining++;\n'
            '          }\n'
            '          _activeConnectionIds.remove(part.task.taskId);\n'
            '          part.launched = false;\n'
            '        }\n'
            '\n'
            '        bool started;\n'
            '        try {\n'
            '          started = await startPart(part.task, part.progress, part.size);\n'
            '        } catch (_) {\n'
            '          // The task was never handed to native IO. This is a local enqueue\n'
            '          // failure, not evidence that the origin cannot sustain the current\n'
            '          // connection level. Restoring/capping slow-start here can pin the\n'
            '          // session at its already-active connection count and silently prevent\n'
            '          // this Range from ever being retried.\n'
            '          rollbackUnownedReservation();\n'
            '          _schedulePartRecovery(session, part);\n'
            '          try {\n'
            '            await _status(session, TaskStatus.running);\n'
            '          } catch (_) {}\n'
            '          return true;\n'
            '        }\n'
            '        if (!started) {\n'
            '          // A false enqueue result has identical ownership semantics: no native\n'
            '          // worker exists, so restore the scheduler reservation and retry the\n'
            '          // exact same taskId/Range without teaching a lower host ceiling.\n'
            '          rollbackUnownedReservation();\n'
            '          _schedulePartRecovery(session, part);\n'
            '          await _status(session, TaskStatus.running);\n'
            '          return true;\n'
            '        }\n'
            '\n'
            '        _armTailStallWatch(session, part);\n'
            '\n'
            '        // On process recovery a child may already be owned by native IO, so a\n'
            '        // fresh running callback is not guaranteed.\n'
            '        if (record != null &&\n'
            '            (record.status == TaskStatus.running ||\n'
            '                record.status == TaskStatus.waitingToRetry)) {\n'
            '          session.currentBatchPendingIds.remove(part.task.taskId);\n'
            '        }\n'
            '      }\n'
            '\n'
            '      if (session.currentBatchRemaining > 0) return true;\n'
            '      if (session.currentBatchPendingIds.isNotEmpty) return true;\n'
            '    }\n'
            '    return true;\n'
            '  }\n'
            '\n'
            '  void _markConnectionReady(_ParallelSession session, _DownloadPart part) {\n'
            '    if (!session.currentBatchPendingIds.remove(part.task.taskId)) return;\n'
            '    if (session.currentBatchRemaining == 0 &&\n'
            '        session.currentBatchPendingIds.isEmpty) {\n'
            '      if (!session.slowStartComplete) {\n'
            '        final active = _activeConnectionsForSession(session);\n'
            '        if (active > session.lastHealthyConnections) {\n'
            '          session.lastHealthyConnections = active;\n'
            '        }\n'
            '        if (session.rampBatchIndex >= session.rampBatches.length) {\n'
            '          session.slowStartComplete = true;\n'
            '        }\n'
            '      }\n'
            '      _schedulePumpAll();\n'
            '    }\n'
            '  }\n'
            '\n'
            '  void _cancelTailStallWatch(_DownloadPart part) {\n'
            '    part.tailStallTimer?.cancel();\n'
            '    part.tailStallTimer = null;\n'
            '  }\n'
            '\n'
            '  void _releaseConnection(_DownloadPart part) {\n'
            '    part.recoveryTimer?.cancel();\n'
            '    part.recoveryTimer = null;\n'
            '    _cancelTailStallWatch(part);\n'
            '    _activeConnectionIds.remove(part.task.taskId);\n'
            '    part.launched = false;\n'
            '    part.speed = 0;\n'
            '    _schedulePumpAll();\n'
            '  }\n'
            '\n'
            '  void _armTailStallWatch(_ParallelSession session, _DownloadPart part) {\n'
            '    if (_disposed ||\n'
            '        !session.active ||\n'
            '        session.deleted ||\n'
            '        part.complete ||\n'
            '        !part.launched ||\n'
            '        part.progress < kParallelNativeCompletionSentinel ||\n'
            '        part.progress >= 1) {\n'
            '      _cancelTailStallWatch(part);\n'
            '      return;\n'
            '    }\n'
            '\n'
            '    final observedProgress = part.credibleProgress;\n'
            '    if (part.tailStallTimer != null &&\n'
            '        observedProgress <= part.tailWatchProgress) {\n'
            '      return;\n'
            '    }\n'
            '\n'
            '    _cancelTailStallWatch(part);\n'
            '    part.tailWatchProgress = observedProgress;\n'
            '    final generation = session.generation;\n'
            '    part.tailStallTimer = Timer(tailStallDelay, () {\n'
            '      part.tailStallTimer = null;\n'
            '      unawaited(\n'
            '        session.serialize(() async {\n'
            '          if (_disposed ||\n'
            '              !session.active ||\n'
            '              session.deleted ||\n'
            '              session.generation != generation ||\n'
            '              part.complete ||\n'
            '              !part.launched) {\n'
            '            return;\n'
            '          }\n'
            '          if (part.credibleProgress > observedProgress) {\n'
            '            _armTailStallWatch(session, part);\n'
            '            return;\n'
            '          }\n'
            '          await _recoverTailStall(session, part);\n'
            '        }),\n'
            '      );\n'
            '    });\n'
            '  }\n'
            '\n'
            '  Future<void> _afterAdoptedPart(_ParallelSession session) async {\n'
            '    await _persist(session);\n'
            '    if (session.parts.every((child) => child.complete)) {\n'
            '      await _assemble(session);\n'
            '    } else {\n'
            '      _scheduleAggregateProgress(session);\n'
            '      _schedulePumpAll();\n'
            '    }\n'
            '  }\n'
            '\n'
            '  /// A tail child that is still marked live but has made no forward progress\n'
            '  /// must not own the final slot forever. First settle/pause the same child and\n'
            '  /// preserve its native resume data. If that exact worker returns but remains\n'
            '  /// stuck, recycle only that immutable Range; all completed sibling ranges are\n'
            '  /// left untouched.\n'
            '  Future<void> _recoverTailStall(\n'
            '    _ParallelSession session,\n'
            '    _DownloadPart part,\n'
            '  ) async {\n'
            '    if (_disposed || !session.active || session.deleted || part.complete)\n'
            '      return;\n'
            '\n'
            '    if (await _adoptExactSizePart(\n'
            '      session,\n'
            '      part,\n'
            '      settleNativeOwner: part.launched,\n'
            '    )) {\n'
            '      await _afterAdoptedPart(session);\n'
            '      return;\n'
            '    }\n'
            '\n'
            '    if (!part.tailRecoveryAttempted) {\n'
            '      part.tailRecoveryAttempted = true;\n'
            '      try {\n'
            '        await pausePart(part.task);\n'
            '      } catch (_) {\n'
            '        // The worker can already be a stale URLSession bookkeeping entry.\n'
            '      }\n'
            '\n'
            '      if (await _adoptExactSizePart(session, part, settleNativeOwner: false)) {\n'
            '        await _afterAdoptedPart(session);\n'
            '        return;\n'
            '      }\n'
            '\n'
            '      _stabilizeSessionForRecovery(session);\n'
            '      _schedulePartRecovery(session, part);\n'
            '      await _persist(session);\n'
            '      await _status(session, TaskStatus.running);\n'
            '      return;\n'
            '    }\n'
            '\n'
            '    await _recycleStalledTailRange(session, part);\n'
            '  }\n'
            '\n'
            '  Future<void> _recycleStalledTailRange(\n'
            '    _ParallelSession session,\n'
            '    _DownloadPart part,\n'
            '  ) async {\n'
            '    final destinationPath = await part.task.filePath();\n'
            '    final partial = await canonicalizePartialDownloadFile(\n'
            '      destinationPath: destinationPath,\n'
            '    );\n'
            '    var savedBytes = partial?.bytes ?? 0;\n'
            '    File? backup;\n'
            '\n'
            '    // cancelTasksWithIds may remove URLSession\'s child file. Preserve a visible\n'
            '    // prefix outside the plugin\'s temp names, then restore it under the same\n'
            '    // child path so DownloadRangeTransfer can append only the missing suffix.\n'
            '    if (savedBytes > 0 && savedBytes < part.size && partial != null) {\n'
            '      backup = File(\'$destinationPath.aw-tail-recovery\');\n'
            '      try {\n'
            '        if (await backup.exists()) await backup.delete();\n'
            '        await partial.file.copy(backup.path);\n'
            '      } catch (_) {\n'
            '        backup = null;\n'
            '      }\n'
            '    } else if (savedBytes > part.size) {\n'
            '      // An oversized child is not a safe prefix of its immutable Range.\n'
            '      savedBytes = 0;\n'
            '    }\n'
            '\n'
            '    _cancelTailStallWatch(part);\n'
            '    _activeConnectionIds.remove(part.task.taskId);\n'
            '    part.launched = false;\n'
            '    part.speed = 0;\n'
            '    session.currentBatchPendingIds.remove(part.task.taskId);\n'
            '\n'
            '    try {\n'
            '      await cancelParts(<String>[part.task.taskId]);\n'
            '    } catch (_) {\n'
            '      // Recovery remains safe even if native already forgot this child.\n'
            '    }\n'
            '\n'
            '    if (backup != null) {\n'
            '      try {\n'
            '        if (await backup.exists()) {\n'
            '          final destination = File(destinationPath);\n'
            '          await destination.parent.create(recursive: true);\n'
            '          await backup.copy(destination.path);\n'
            '          await backup.delete();\n'
            '        }\n'
            '      } catch (_) {}\n'
            '    }\n'
            '\n'
            '    final restored = await canonicalizePartialDownloadFile(\n'
            '      destinationPath: destinationPath,\n'
            '    );\n'
            '    savedBytes = restored?.bytes ?? 0;\n'
            '    if (savedBytes == part.size && part.size > 0) {\n'
            '      part.complete = true;\n'
            '      part.progress = 1;\n'
            '      part.credibleProgress = 1;\n'
            '      await saveRecord(\n'
            '        TaskRecord(part.task, TaskStatus.complete, 1, part.size),\n'
            '      );\n'
            '      onPartProgress(session.task.taskId, part.task.taskId, 1);\n'
            '      await _afterAdoptedPart(session);\n'
            '      return;\n'
            '    }\n'
            '    if (savedBytes < 0 || savedBytes > part.size) savedBytes = 0;\n'
            '\n'
            '    // If native resume data/temp bytes are genuinely inaccessible, zero means\n'
            '    // re-fetch only this one immutable Range. Never restart the parent episode\n'
            '    // or any of its already-completed siblings.\n'
            '    part.progress = part.size > 0 ? savedBytes / part.size : 0;\n'
            '    part.credibleProgress = part.progress;\n'
            '    part.recoveryAttempts = 0;\n'
            '    await saveRecord(\n'
            '      TaskRecord(part.task, TaskStatus.paused, part.progress, part.size),\n'
            '    );\n'
            '    onPartProgress(\n'
            '      session.task.taskId,\n'
            '      part.task.taskId,\n'
            '      part.credibleProgress,\n'
            '    );\n'
            '    await _persist(session);\n'
            '    _stabilizeSessionForRecovery(session);\n'
            '    _schedulePartRecovery(session, part);\n'
            '    await _status(session, TaskStatus.running);\n'
            '  }\n'
            '\n'
            '  /// Adopt a range whose complete callback was lost after every requested byte\n'
            '  /// already reached the destination file. If native still owns the task, ask\n'
            '  /// it to pause/finalize first; then verify the exact byte count again before\n'
            '  /// changing durable state. Never infer completion from 99% alone.\n'
            '  Future<bool> _adoptExactSizePart(\n'
            '    _ParallelSession session,\n'
            '    _DownloadPart part, {\n'
            '    required bool settleNativeOwner,\n'
            '  }) async {\n'
            '    if (part.complete || session.deleted) return part.complete;\n'
            '\n'
            '    final path = await part.task.filePath();\n'
            '    final file = File(path);\n'
            '    if (!await file.exists() || await file.length() != part.size) return false;\n'
            '\n'
            '    if (settleNativeOwner) {\n'
            '      try {\n'
            '        await pausePart(part.task);\n'
            '      } catch (_) {\n'
            '        // A task that has already finished natively may no longer be pausable.\n'
            '        // The second exact-size verification below remains the source of truth.\n'
            '      }\n'
            '    }\n'
            '\n'
            '    if (!await file.exists() || await file.length() != part.size) return false;\n'
            '\n'
            '    _markConnectionReady(session, part);\n'
            '    _releaseConnection(part);\n'
            '    part.complete = true;\n'
            '    part.progress = 1;\n'
            '    part.credibleProgress = 1;\n'
            '    await saveRecord(TaskRecord(part.task, TaskStatus.complete, 1, part.size));\n'
            '    onPartProgress(session.task.taskId, part.task.taskId, 1);\n'
            '    return true;\n'
            '  }\n'
            '\n'
            '  void _capSessionAt(_ParallelSession session, int ceiling) {\n'
            '    final safe = ceiling\n'
            '        .clamp(kDownloadPartsMin, kDownloadGlobalConnectionBudget)\n'
            '        .toInt();\n'
            '    if (safe >= session.connectionCeiling) return;\n'
            '    session.connectionCeiling = safe;\n'
            '    session.slowStartComplete = true;\n'
            '    session.rampBatches = const <int>[];\n'
            '    session.rampBatchIndex = 0;\n'
            '    // Do not clear pending IDs: already-launched native children must prove\n'
            '    // they recovered before the scheduler considers opening replacements.\n'
            '    session.currentBatchRemaining = 0;\n'
            '  }\n'
            '\n'
            '  /// Any automatic child recovery freezes slow-start at the last proven\n'
            '  /// healthy level. This prevents one interrupted batch from immediately\n'
            '  /// expanding again while still allowing spare tail work to replace the\n'
            '  /// backing-off connection at that safe ceiling.\n'
            '  void _stabilizeSessionForRecovery(_ParallelSession session) {\n'
            '    final active = _activeConnectionsForSession(session);\n'
            '    final safe = session.lastHealthyConnections > 0\n'
            '        ? session.lastHealthyConnections\n'
            '        : (active > 0 ? active : kDownloadPartsMin);\n'
            '    _capSessionAt(session, safe);\n'
            '  }\n'
            '\n'
            '  void _applyConnectionPressure(\n'
            '    _ParallelSession session,\n'
            '    TaskStatusUpdate update,\n'
            '  ) {\n'
            '    final pressure = downloadConnectionPressureFor(update);\n'
            '    if (pressure == DownloadConnectionPressure.none) return;\n'
            '\n'
            '    final active = _activeConnectionsForSession(session).clamp(1, 1 << 30);\n'
            '    final growthInFlight =\n'
            '        session.currentBatchPendingIds.isNotEmpty ||\n'
            '        session.currentBatchRemaining > 0;\n'
            '    var fallback = growthInFlight && session.lastHealthyConnections > 0\n'
            '        ? session.lastHealthyConnections\n'
            '        : (active <= 1 ? 1 : (active + 1) ~/ 2);\n'
            '    if (fallback > session.connectionCeiling) {\n'
            '      fallback = session.connectionCeiling;\n'
            '    }\n'
            '\n'
            '    if (pressure == DownloadConnectionPressure.host) {\n'
            '      final learned = _connectionGovernor.learnHostCeiling(\n'
            '        session.task.url,\n'
            '        fallback,\n'
            '      );\n'
            '      onHostPressure?.call(session.task.url, learned);\n'
            '      for (final sibling in _sessions.values) {\n'
            '        if (!sibling.active || sibling.deleted) continue;\n'
            '        if (_connectionGovernor.sameOrigin(\n'
            '          sibling.task.url,\n'
            '          session.task.url,\n'
            '        )) {\n'
            '          _capSessionAt(sibling, learned);\n'
            '        }\n'
            '      }\n'
            '    } else {\n'
            '      final learned = _connectionGovernor.learnTransferCeiling(\n'
            '        session.task.url,\n'
            '        fallback,\n'
            '      );\n'
            '      _capSessionAt(session, learned);\n'
            '    }\n'
            '  }\n'
            '\n'
            '  void _scheduleDiskProgressPoll(_ParallelSession session) {\n'
            '    if (_disposed || !session.active || session.deleted) return;\n'
            '    if (session.diskProgressTimer != null) return;\n'
            '\n'
            '    if (session.lastDiskObservedBytes < 0) {\n'
            '      session.lastDiskObservedBytes = session.creditedBytes;\n'
            '      session.lastDiskObservedAt = DateTime.now();\n'
            '    }\n'
            '    session.diskProgressTimer = Timer(diskProgressPollInterval, () {\n'
            '      session.diskProgressTimer = null;\n'
            '      if (_disposed) return;\n'
            '      unawaited(\n'
            '        session.serialize(() async {\n'
            '          if (_disposed || !session.active || session.deleted) return;\n'
            '          try {\n'
            '            await _pollDiskProgress(session);\n'
            '          } catch (_) {\n'
            '            // This is only a fallback for missing native callbacks. A transient\n'
            '            // file move/read race must never pause a healthy parent download.\n'
            '          } finally {\n'
            '            if (!_disposed && session.active && !session.deleted) {\n'
            '              _scheduleDiskProgressPoll(session);\n'
            '            }\n'
            '          }\n'
            '        }),\n'
            '      );\n'
            '    });\n'
            '  }\n'
            '\n'
            '  Future<void> _pollDiskProgress(_ParallelSession session) async {\n'
            '    var changed = false;\n'
            '\n'
            '    for (final part in session.parts) {\n'
            '      if (part.complete) continue;\n'
            '      try {\n'
            '        final file = File(await part.task.filePath());\n'
            '        if (!await file.exists()) continue;\n'
            '        final bytes = await file.length();\n'
            '        if (bytes <= 0 || bytes > part.size) continue;\n'
            '\n'
            '        if (bytes == part.size &&\n'
            '            await _adoptExactSizePart(\n'
            '              session,\n'
            '              part,\n'
            '              settleNativeOwner: part.launched,\n'
            '            )) {\n'
            '          changed = true;\n'
            '          continue;\n'
            '        }\n'
            '\n'
            '        final diskProgress = (bytes / part.size).clamp(0.0, 1.0).toDouble();\n'
            '        if (diskProgress <= part.credibleProgress) continue;\n'
            '\n'
            '        part.credibleProgress = diskProgress;\n'
            '        if (part.progress >= kParallelNativeCompletionSentinel ||\n'
            '            diskProgress > part.progress) {\n'
            '          part.progress = diskProgress;\n'
            '        }\n'
            '        part.recoveryAttempts = 0;\n'
            '        part.tailRecoveryAttempted = false;\n'
            '        if (part.launched) _markConnectionReady(session, part);\n'
            '        onPartProgress(\n'
            '          session.task.taskId,\n'
            '          part.task.taskId,\n'
            '          part.credibleProgress,\n'
            '        );\n'
            '        changed = true;\n'
            '      } on FileSystemException {\n'
            '        // URLSession may atomically move a child while it is sampled.\n'
            '      }\n'
            '    }\n'
            '\n'
            '    final now = DateTime.now();\n'
            '    final creditedBytes = session.creditedBytes;\n'
            '    final previousBytes = session.lastDiskObservedBytes;\n'
            '    final previousAt = session.lastDiskObservedAt;\n'
            '    if (previousBytes >= 0 && previousAt != null) {\n'
            '      final elapsedMicros = now.difference(previousAt).inMicroseconds;\n'
            '      final deltaBytes = creditedBytes - previousBytes;\n'
            '      if (elapsedMicros > 0 && deltaBytes > 0) {\n'
            '        session.diskObservedSpeed =\n'
            '            deltaBytes *\n'
            '            Duration.microsecondsPerSecond /\n'
            '            elapsedMicros /\n'
            '            1000 /\n'
            '            1000;\n'
            '      } else if (deltaBytes <= 0) {\n'
            '        session.diskObservedSpeed = 0;\n'
            '      }\n'
            '    }\n'
            '    session.lastDiskObservedBytes = creditedBytes;\n'
            '    session.lastDiskObservedAt = now;\n'
            '\n'
            '    if (!changed) return;\n'
            '    _scheduleProgressPersist(session);\n'
            '    if (session.parts.every((part) => part.complete)) {\n'
            '      await _assemble(session);\n'
            '      return;\n'
            '    }\n'
            '    if (!session.parentRunningReported) {\n'
            '      await _status(session, TaskStatus.running);\n'
            '    }\n'
            '    _scheduleAggregateProgress(session);\n'
            '    _schedulePumpAll();\n'
            '  }\n'
            '\n'
            '  void _scheduleProgressPersist(_ParallelSession session) {\n'
            '    if (_disposed || !session.active || session.deleted) return;\n'
            '    session.progressPersistDirty = true;\n'
            '    if (session.progressPersistTimer != null) return;\n'
            '\n'
            '    session.progressPersistTimer = Timer(kParallelProgressPersistInterval, () {\n'
            '      session.progressPersistTimer = null;\n'
            '      if (_disposed) return;\n'
            '      unawaited(\n'
            '        session.serialize(() async {\n'
            '          if (_disposed ||\n'
            '              !session.active ||\n'
            '              session.deleted ||\n'
            '              !session.progressPersistDirty) {\n'
            '            return;\n'
            '          }\n'
            '          session.progressPersistDirty = false;\n'
            '          await _persist(session);\n'
            '        }),\n'
            '      );\n'
            '    });\n'
            '  }\n'
            '\n'
            '  Future<void> _writeParentRecord(_ParallelSession session, TaskRecord record) {\n'
            '    late final Future<void> operation;\n'
            '    operation = session.parentRecordWrite.then((_) => saveRecord(record));\n'
            '    session.parentRecordWrite = operation.then<void>(\n'
            '      (_) {},\n'
            '      onError: (Object error, StackTrace stackTrace) {\n'
            '        diagnosticLog?.record(\'parallel.parentRecordPersistFailed\', {\n'
            '          \'taskId\': session.task.taskId,\n'
            '          \'errorType\': error.runtimeType.toString(),\n'
            '        });\n'
            '      },\n'
            '    );\n'
            '    return operation;\n'
            '  }\n'
            '\n'
            '  void _scheduleAggregateProgress(_ParallelSession session) {\n'
            '    if (_disposed || !session.active || session.deleted) return;\n'
            '    session.aggregateProgressDirty = true;\n'
            '    if (session.aggregateProgressTimer != null) return;\n'
            '\n'
            '    // Presentation must not wait behind manifest fsync/recovery bookkeeping.\n'
            '    // Snapshot on the one-second clock and publish immediately; parent DB\n'
            '    // writes use their own ordered chain so a stale running write can never\n'
            '    // overwrite a later pause/complete record.\n'
            '    session.aggregateProgressTimer = Timer(kParallelProgressCoalesceDelay, () {\n'
            '      session.aggregateProgressTimer = null;\n'
            '      if (_disposed ||\n'
            '          !session.active ||\n'
            '          session.deleted ||\n'
            '          !session.aggregateProgressDirty) {\n'
            '        return;\n'
            '      }\n'
            '      session.aggregateProgressDirty = false;\n'
            '      _emitAggregateProgress(session);\n'
            '    });\n'
            '  }\n'
            '\n'
            '  void _emitAggregateProgress(_ParallelSession session) {\n'
            '    if (_disposed || !session.active || session.deleted) return;\n'
            '    final task = session.task;\n'
            '    final progress = session.progress;\n'
            '    final creditedBytes = session.creditedBytes;\n'
            '    final expectedBytes = session.size;\n'
            '    final telemetry = _speedTelemetry.observe(\n'
            '      taskId: task.taskId,\n'
            '      transferredBytes: creditedBytes,\n'
            '      expectedBytes: expectedBytes,\n'
            '    );\n'
            '    final speed = telemetry.speedBytesPerSecond > 0\n'
            '        ? telemetry.speedBytesPerSecond / 1000 / 1000\n'
            '        : 0.0;\n'
            '    final timeRemaining = telemetry.timeRemaining;\n'
            '\n'
            '    if (speed > 0) {\n'
            '      final now = DateTime.now();\n'
            '      final previousSample = session.lastHostProfileSampleAt;\n'
            '      if (previousSample == null ||\n'
            '          now.difference(previousSample) >=\n'
            '              kParallelHostProfileSampleInterval) {\n'
            '        session.lastHostProfileSampleAt = now;\n'
            '        onHostSample?.call(\n'
            '          task.url,\n'
            '          _activeConnectionsForSession(session).clamp(1, 1 << 30),\n'
            '          speed * 1000 * 1000,\n'
            '        );\n'
            '      }\n'
            '    }\n'
            '\n'
            '    onUpdate(\n'
            '      TaskProgressUpdate(task, progress, expectedBytes, speed, timeRemaining),\n'
            '    );\n'
            '    unawaited(\n'
            '      _writeParentRecord(\n'
            '        session,\n'
            '        TaskRecord(task, TaskStatus.running, progress, expectedBytes),\n'
            '      ).catchError((Object _, StackTrace __) {}),\n'
            '    );\n'
            '  }\n'
            '\n'
            '  Duration _aggregateTimeRemaining(_ParallelSession session, double speedMb) {\n'
            '    if (speedMb <= 0 || session.progress >= 1) {\n'
            '      return const Duration(seconds: -1);\n'
            '    }\n'
            '    final remainingBytes = session.parts.fold<double>(\n'
            '      0,\n'
            '      (sum, part) => sum + part.size * (1 - part.credibleProgress),\n'
            '    );\n'
            '    if (remainingBytes <= 0) return Duration.zero;\n'
            '\n'
            '    final seconds = (remainingBytes / (speedMb * 1000 * 1000)).ceil();\n'
            '    return Duration(seconds: seconds < 1 ? 1 : seconds);\n'
            '  }\n'
            '\n'
            '  void _scheduleCoordinatorRecovery(_ParallelSession session) {\n'
            '    if (_disposed || !session.active || session.deleted) return;\n'
            '    if (session.coordinatorRecoveryTimer != null) return;\n'
            '    final generation = session.generation;\n'
            '    session.coordinatorRecoveryTimer = Timer(recoveryDelay, () {\n'
            '      session.coordinatorRecoveryTimer = null;\n'
            '      if (_disposed) return;\n'
            '      unawaited(\n'
            '        session.serialize(() async {\n'
            '          if (_disposed ||\n'
            '              !session.active ||\n'
            '              session.deleted ||\n'
            '              session.generation != generation) {\n'
            '            return;\n'
            '          }\n'
            '          try {\n'
            '            await _restoreNativeOwnership(session);\n'
            '          } catch (_) {}\n'
            '          _scheduleDiskProgressPoll(session);\n'
            '          try {\n'
            '            if (!session.parentRunningReported) {\n'
            '              await _status(session, TaskStatus.running);\n'
            '            }\n'
            '          } catch (_) {}\n'
            '          try {\n'
            '            await _pumpSession(session);\n'
            '          } catch (_) {}\n'
            '          try {\n'
            '            await _persist(session);\n'
            '          } catch (_) {}\n'
            '          _schedulePumpAll();\n'
            '        }),\n'
            '      );\n'
            '    });\n'
            '  }\n'
            '\n'
            '  void _schedulePumpAll() {\n'
            '    if (_disposed || _pumpFuture != null) return;\n'
            '\n'
            '    late final Future<void> pump;\n'
            '    pump =\n'
            '        Future<void>.microtask(() async {\n'
            '              final sessions = List<_ParallelSession>.from(_sessions.values);\n'
            '              for (final session in sessions) {\n'
            '                if (_disposed) return;\n'
            '                if (!session.active || session.deleted) continue;\n'
            '                await session.serialize(() async {\n'
            '                  if (_disposed || !session.active || session.deleted) return;\n'
            '                  try {\n'
            '                    if (!await _pumpSession(session)) {\n'
            '                      _scheduleCoordinatorRecovery(session);\n'
            '                    } else {\n'
            '                      await _persist(session);\n'
            '                    }\n'
            '                  } catch (_) {\n'
            '                    // Coordinator bookkeeping is not a user-visible pause.\n'
            '                    // Keep native owners untouched and reconcile them shortly.\n'
            '                    _scheduleCoordinatorRecovery(session);\n'
            '                  }\n'
            '                });\n'
            '              }\n'
            '            })\n'
            '            .catchError((Object _, StackTrace _) {\n'
            '              // Session-level failures park their parent. An unexpected lifecycle race\n'
            '              // must not become an unhandled asynchronous exception.\n'
            '            })\n'
            '            .whenComplete(() {\n'
            '              if (identical(_pumpFuture, pump)) _pumpFuture = null;\n'
            '              if (!_disposed &&\n'
            '                  _activeConnectionIds.length < _connectionBudget &&\n'
            '                  _sessions.values.any(_hasImmediatelyPumpableWork)) {\n'
            '                _schedulePumpAll();\n'
            '              }\n'
            '            });\n'
            '    _pumpFuture = pump;\n'
            '  }\n'
            '\n'
            '  bool _hasImmediatelyPumpableWork(_ParallelSession session) {\n'
            '    if (_disposed || !session.active || session.deleted) return false;\n'
            '    if (_activeConnectionsForSession(session) >= session.connectionCeiling) {\n'
            '      return false;\n'
            '    }\n'
            '    final launchable = _launchablePartCount(session);\n'
            '    if (launchable <= 0) return false;\n'
            '    if (session.currentBatchRemaining > 0) return true;\n'
            '    if (session.currentBatchPendingIds.isNotEmpty) return false;\n'
            '    if (session.rampBatchIndex < session.rampBatches.length) return true;\n'
            '    return launchable > 0;\n'
            '  }\n'
            '\n'
            '  /// iOS writes DownloadTask bodies into URLSession-owned temporary files, so\n'
            '  /// the final `.part` path can remain invisible until didFinishDownloadingTo.\n'
            '  /// The native delegate bridge reports the bytes here while they are still in\n'
            '  /// that temp file. This is byte evidence, not a guessed percentage: it wakes\n'
            '  /// the logical parent, advances slow-start and keeps speed/progress live even\n'
            '  /// when background_downloader\'s Dart callbacks are delayed or lost.\n'
            '  Future<void> handleNativeChunkUpdate({\n'
            '    required String parentTaskId,\n'
            '    required String chunkTaskId,\n'
            '    double? progress,\n'
            '    int? writtenBytes,\n'
            '    int? expectedBytes,\n'
            '    double? speedBytesPerSecond,\n'
            '    bool completed = false,\n'
            '  }) async {\n'
            '    if (_disposed) return;\n'
            '    final session = _sessions[parentTaskId] ?? _children[chunkTaskId];\n'
            '    if (session == null || session.deleted) return;\n'
            '\n'
            '    await session.serialize(() async {\n'
            '      if (_disposed ||\n'
            '          session.deleted ||\n'
            '          !identical(_sessions[session.task.taskId], session)) {\n'
            '        return;\n'
            '      }\n'
            '\n'
            '      _DownloadPart? part;\n'
            '      for (final candidate in session.parts) {\n'
            '        if (candidate.task.taskId == chunkTaskId) {\n'
            '          part = candidate;\n'
            '          break;\n'
            '        }\n'
            '      }\n'
            '      if (part == null || part.complete) return;\n'
            '\n'
            '      // The immutable Range in the manifest is the authority. Never trust a\n'
            '      // server-reported expected length enough to credit bytes outside it.\n'
            '      int? observedBytes;\n'
            '      double? credible;\n'
            '      if (writtenBytes != null &&\n'
            '          writtenBytes >= 0 &&\n'
            '          writtenBytes <= part.size) {\n'
            '        observedBytes = writtenBytes;\n'
            '        credible = part.size > 0 ? writtenBytes / part.size : 0;\n'
            '      } else if (progress != null && progress >= 0 && progress <= 1) {\n'
            '        credible = progress;\n'
            '      }\n'
            '\n'
            '      // didFinishDownloadingTo is hooked after the plugin moves the temp body\n'
            '      // to our final child path. Completion still requires exact local bytes.\n'
            '      if (completed) {\n'
            '        final file = File(await part.task.filePath());\n'
            '        if (await file.exists() &&\n'
            '            await file.length() == part.size &&\n'
            '            await _adoptExactSizePart(\n'
            '              session,\n'
            '              part,\n'
            '              settleNativeOwner: false,\n'
            '            )) {\n'
            '          await _afterAdoptedPart(session);\n'
            '          return;\n'
            '        }\n'
            '      }\n'
            '\n'
            '      if (credible == null) return;\n'
            '      credible = credible.clamp(0.0, 1.0).toDouble();\n'
            '\n'
            '      final previousCredible = part.credibleProgress;\n'
            '      final now = DateTime.now();\n'
            '      if (credible > previousCredible) {\n'
            '        if (speedBytesPerSecond != null && speedBytesPerSecond > 0) {\n'
            '          part.speed = speedBytesPerSecond / 1000 / 1000;\n'
            '        } else if (observedBytes != null &&\n'
            '            part.lastNativeBridgeBytes >= 0 &&\n'
            '            part.lastNativeBridgeAt != null) {\n'
            '          final elapsedMicros = now\n'
            '              .difference(part.lastNativeBridgeAt!)\n'
            '              .inMicroseconds;\n'
            '          final deltaBytes = observedBytes - part.lastNativeBridgeBytes;\n'
            '          if (elapsedMicros > 0 && deltaBytes > 0) {\n'
            '            part.speed =\n'
            '                deltaBytes *\n'
            '                Duration.microsecondsPerSecond /\n'
            '                elapsedMicros /\n'
            '                1000 /\n'
            '                1000;\n'
            '          }\n'
            '        }\n'
            '\n'
            '        part.credibleProgress = credible;\n'
            '        if (part.progress >= kParallelNativeCompletionSentinel ||\n'
            '            credible > part.progress) {\n'
            '          part.progress = credible;\n'
            '        }\n'
            '        part.recoveryAttempts = 0;\n'
            '        part.tailRecoveryAttempted = false;\n'
            '      } else if (speedBytesPerSecond != null && speedBytesPerSecond > 0) {\n'
            '        part.speed = speedBytesPerSecond / 1000 / 1000;\n'
            '      }\n'
            '\n'
            '      if (observedBytes != null) {\n'
            '        part.lastNativeBridgeBytes = observedBytes;\n'
            '        part.lastNativeBridgeAt = now;\n'
            '      }\n'
            '\n'
            '      if (session.active) {\n'
            '        part.launched = true;\n'
            '        _activeConnectionIds.add(part.task.taskId);\n'
            '        _markConnectionReady(session, part);\n'
            '      }\n'
            '\n'
            '      if (credible > previousCredible || completed) {\n'
            '        onPartProgress(\n'
            '          session.task.taskId,\n'
            '          part.task.taskId,\n'
            '          part.credibleProgress,\n'
            '        );\n'
            '        _scheduleProgressPersist(session);\n'
            '      }\n'
            '\n'
            '      if (!session.active) return;\n'
            '      if (!session.parentRunningReported) {\n'
            '        await _status(session, TaskStatus.running);\n'
            '      }\n'
            '      _scheduleAggregateProgress(session);\n'
            '      _schedulePumpAll();\n'
            '    });\n'
            '  }\n'
            '\n'
            '  bool handleUpdate(TaskUpdate update) {\n'
            '    if (update.task.group != kPersistentDownloadChunkGroup) return false;\n'
            '    if (_disposed) return true;\n'
            '    final session = _children[update.task.taskId];\n'
            '    if (session == null) return true;\n'
            '    final generation = session.generation;\n'
            '\n'
            '    unawaited(\n'
            '      session.serialize(() async {\n'
            '        try {\n'
            '          if (_disposed ||\n'
            '              session.deleted ||\n'
            '              !identical(_sessions[session.task.taskId], session))\n'
            '            return;\n'
            '          final part = session.parts.firstWhere(\n'
            '            (part) => part.task.taskId == update.task.taskId,\n'
            '          );\n'
            '          // Completion is durable; a late running/progress/retry callback must\n'
            '          // never reserve its connection again or park the remaining parts.\n'
            '          if (part.complete) return;\n'
            '          if (generation != session.generation &&\n'
            '              !(update is TaskStatusUpdate &&\n'
            '                  update.status == TaskStatus.complete))\n'
            '            return;\n'
            '\n'
            '          if (update is TaskProgressUpdate &&\n'
            '              update.progress >= 0 &&\n'
            '              update.progress <= 1) {\n'
            '            // On iOS the URLSession bridge carries exact byte counts from the\n'
            '            // temp file. background_downloader emits a second progress callback\n'
            '            // for the same child. Ignore that duplicate while the byte bridge\n'
            '            // is healthy; if the bridge goes quiet, plugin progress becomes the\n'
            '            // fallback automatically.\n'
            '            final nativeBridgeFresh =\n'
            '                part.lastNativeBridgeAt != null &&\n'
            '                DateTime.now().difference(part.lastNativeBridgeAt!) <\n'
            '                    const Duration(seconds: 2);\n'
            '            if (nativeBridgeFresh) return;\n'
            '\n'
            '            if (session.active) {\n'
            '              part.launched = true;\n'
            '              _activeConnectionIds.add(part.task.taskId);\n'
            '              _markConnectionReady(session, part);\n'
            '            }\n'
            '            final incoming = update.progress;\n'
            '            final previousCredibleProgress = part.credibleProgress;\n'
            '\n'
            '            if (incoming >= kParallelNativeCompletionSentinel) {\n'
            '              // 0.999 (and a progress-only 1.0) is a native completion\n'
            '              // sentinel. Keep it for exact-size adoption/watchdog recovery,\n'
            '              // but do not count it as downloaded bytes.\n'
            '              if (incoming > part.progress) part.progress = incoming;\n'
            '            } else if (incoming > part.credibleProgress) {\n'
            '              // A normal sample is byte-credible. If this connection is\n'
            '              // recovering from a previously observed 0.999 sentinel, allow\n'
            '              // the raw marker to move back below 0.999 so the tail watchdog\n'
            '              // is canceled by genuine forward progress.\n'
            '              part.credibleProgress = incoming;\n'
            '              if (part.progress >= kParallelNativeCompletionSentinel ||\n'
            '                  incoming > part.progress) {\n'
            '                part.progress = incoming;\n'
            '              }\n'
            '            }\n'
            '\n'
            '            if (part.credibleProgress > previousCredibleProgress) {\n'
            '              // A recovered connection that actually writes bytes is healthy\n'
            '              // again. Future interruptions start from the short backoff and\n'
            '              // the tail watchdog gets a fresh grace period.\n'
            '              part.recoveryAttempts = 0;\n'
            '              part.tailRecoveryAttempted = false;\n'
            '            }\n'
            '            part.speed = update.networkSpeed > 0 ? update.networkSpeed : 0;\n'
            '\n'
            '            // URLSession occasionally writes the complete range and reports\n'
            '            // 0.999, then never delivers TaskStatus.complete. Only take over\n'
            '            // when the exact requested byte count is already on disk; pausing\n'
            '            // the child here settles native ownership before assembly deletes\n'
            '            // the .part files.\n'
            '            if (session.active &&\n'
            '                incoming >= kParallelNativeCompletionSentinel &&\n'
            '                await _adoptExactSizePart(\n'
            '                  session,\n'
            '                  part,\n'
            '                  settleNativeOwner: part.launched,\n'
            '                )) {\n'
            '              await _afterAdoptedPart(session);\n'
            '              return;\n'
            '            }\n'
            '\n'
            '            _armTailStallWatch(session, part);\n'
            '            onPartProgress(\n'
            '              session.task.taskId,\n'
            '              part.task.taskId,\n'
            '              part.credibleProgress,\n'
            '            );\n'
            '            _scheduleProgressPersist(session);\n'
            '            if (session.active) {\n'
            '              _scheduleAggregateProgress(session);\n'
            '            }\n'
            '            _schedulePumpAll();\n'
            '            return;\n'
            '          }\n'
            '\n'
            '          if (update is! TaskStatusUpdate) return;\n'
            '\n'
            '          if (session.active && update.status == TaskStatus.waitingToRetry) {\n'
            '            // Only legacy/native workers created before the unified retry\n'
            '            // policy should reach here. Keep their real native ownership\n'
            '            // reserved, but never turn this internal state into a paused parent\n'
            '            // notification. Newly created/restored child definitions use zero\n'
            '            // native retries and recover through the scheduler below.\n'
            '            part.launched = true;\n'
            '            part.speed = 0;\n'
            '            _activeConnectionIds.add(part.task.taskId);\n'
            '            _applyConnectionPressure(session, update);\n'
            '            _armTailStallWatch(session, part);\n'
            '            await _persist(session);\n'
            '            await _status(session, TaskStatus.running);\n'
            '            return;\n'
            '          }\n'
            '\n'
            '          if (update.status == TaskStatus.complete) {\n'
            '            final file = File(await part.task.filePath());\n'
            '            final exists = await file.exists();\n'
            '            final length = exists ? await file.length() : -1;\n'
            '\n'
            '            // Some origins advertise/probe as range-capable, then answer the\n'
            '            // real ranged child with a full HTTP 200. background_downloader\n'
            '            // writes that full body to the child file. Gopeed keeps its resolve\n'
            '            // response as a sequential fallback in this exact situation. Do\n'
            '            // the equivalent here: if the child already contains the complete\n'
            '            // episode, promote those bytes directly instead of throwing them\n'
            '            // away and downloading the file again with one connection.\n'
            '            if (exists &&\n'
            '                length == session.size &&\n'
            '                length != part.size &&\n'
            '                await _adoptIgnoredRangeFullBody(session, part, update)) {\n'
            '              return;\n'
            '            }\n'
            '\n'
            '            if (!exists || length != part.size) {\n'
            '              // A completed callback with the wrong durable byte count is a\n'
            '              // data-integrity boundary, not a coordinator race. Keep the\n'
            '              // bytes for diagnosis/resume and park the parent deterministically.\n'
            '              await _pause(session);\n'
            '              return;\n'
            '            }\n'
            '            if (!await _validateCompletedNativeRange(session, part, update)) {\n'
            '              return;\n'
            '            }\n'
            '            _markConnectionReady(session, part);\n'
            '            _releaseConnection(part);\n'
            '            part.complete = true;\n'
            '            part.progress = 1;\n'
            '            part.credibleProgress = 1;\n'
            '            onPartProgress(session.task.taskId, part.task.taskId, 1);\n'
            '            await _persist(session);\n'
            '            if (session.active &&\n'
            '                session.parts.every((child) => child.complete)) {\n'
            '              await _assemble(session);\n'
            '            } else {\n'
            '              _scheduleAggregateProgress(session);\n'
            '              _schedulePumpAll();\n'
            '            }\n'
            '            return;\n'
            '          }\n'
            '\n'
            '          if (session.active && update.status == TaskStatus.running) {\n'
            '            part.launched = true;\n'
            '            _activeConnectionIds.add(part.task.taskId);\n'
            '            _markConnectionReady(session, part);\n'
            '            _armTailStallWatch(session, part);\n'
            '            await _status(session, TaskStatus.running);\n'
            '            _schedulePumpAll();\n'
            '            return;\n'
            '          }\n'
            '\n'
            '          if (session.active &&\n'
            '              (update.status == TaskStatus.failed ||\n'
            '                  update.status == TaskStatus.notFound ||\n'
            '                  update.status == TaskStatus.canceled ||\n'
            '                  update.status == TaskStatus.paused)) {\n'
            '            if (update.status == TaskStatus.failed) {\n'
            '              _applyConnectionPressure(session, update);\n'
            '            }\n'
            '\n'
            '            if (_shouldAutomaticallyRecoverPart(update)) {\n'
            '              // A native child finishing/pausing is not the same thing as the\n'
            '              // logical episode stopping. Release only this child\'s connection\n'
            '              // during backoff; healthy siblings/tail work keep moving. The\n'
            '              // parent remains running so recovery never emits a fake pause.\n'
            '              _stabilizeSessionForRecovery(session);\n'
            '              _schedulePartRecovery(session, part);\n'
            '              await _status(session, TaskStatus.running);\n'
            '              return;\n'
            '            }\n'
            '\n'
            '            // Permanent client-side HTTP errors (for example 401/403/404) are\n'
            '            // not helped by hammering the same signed URL forever. Preserve all\n'
            '            // bytes and park the logical episode so a later source refresh/user\n'
            '            // resume can obtain a new URL.\n'
            '            _markConnectionReady(session, part);\n'
            '            _releaseConnection(part);\n'
            '            await _pause(session);\n'
            '          }\n'
            '        } catch (_) {\n'
            '          if (!_disposed && session.active && !session.deleted) {\n'
            '            _scheduleCoordinatorRecovery(session);\n'
            '          }\n'
            '        }\n'
            '      }),\n'
            '    );\n'
            '    return true;\n'
            '  }\n'
            '\n'
            '  Future<bool> pause(ParallelDownloadTask task) async {\n'
            '    if (_disposed) return false;\n'
            '    if (!await restore(task)) return false;\n'
            '    final session = _sessions[task.taskId]!;\n'
            '    return session.serialize(() => _pause(session));\n'
            '  }\n'
            '\n'
            '  bool _shouldAutomaticallyRecoverPart(TaskStatusUpdate update) {\n'
            '    switch (update.status) {\n'
            '      case TaskStatus.paused:\n'
            '      case TaskStatus.canceled:\n'
            '      case TaskStatus.notFound:\n'
            '        return true;\n'
            '      case TaskStatus.failed:\n'
            '        final exception = update.exception;\n'
            '        final statusCode =\n'
            '            update.responseStatusCode ??\n'
            '            (exception is TaskHttpException\n'
            '                ? exception.httpResponseCode\n'
            '                : null);\n'
            '        if (statusCode == null) return true;\n'
            '        return statusCode == 408 ||\n'
            '            statusCode == 425 ||\n'
            '            statusCode == 429 ||\n'
            '            (statusCode >= 500 && statusCode <= 599);\n'
            '      case TaskStatus.enqueued:\n'
            '      case TaskStatus.running:\n'
            '      case TaskStatus.complete:\n'
            '      case TaskStatus.waitingToRetry:\n'
            '        return false;\n'
            '    }\n'
            '  }\n'
            '\n'
            '  Duration _partRecoveryDelay(int attempts) {\n'
            '    final multiplier = attempts\n'
            '        .clamp(1, kParallelRecoveryMaxBackoffMultiplier)\n'
            '        .toInt();\n'
            '    return recoveryDelay * multiplier;\n'
            '  }\n'
            '\n'
            '  bool _schedulePartRecovery(_ParallelSession session, _DownloadPart part) {\n'
            '    if (part.recoveryTimer != null) return true;\n'
            '    if (!session.active || session.deleted || _disposed || part.complete) {\n'
            '      return false;\n'
            '    }\n'
            '    _cancelTailStallWatch(part);\n'
            '    part.recoveryAttempts++;\n'
            '    part.speed = 0;\n'
            '\n'
            '    // Backoff is not an active connection. Free the reserved slot immediately\n'
            '    // so another durable range can use it. The failed range becomes launchable\n'
            '    // only after its timer expires and then re-enters _pumpSession, where both\n'
            '    // the per-session ceiling and the global governor are checked again.\n'
            '    part.launched = false;\n'
            '    _activeConnectionIds.remove(part.task.taskId);\n'
            '    session.currentBatchPendingIds.remove(part.task.taskId);\n'
            '    final generation = session.generation;\n'
            '    part.recoveryTimer = Timer(_partRecoveryDelay(part.recoveryAttempts), () {\n'
            '      unawaited(\n'
            '        session.serialize(() async {\n'
            '          if (_disposed ||\n'
            '              !session.active ||\n'
            '              session.deleted ||\n'
            '              session.generation != generation ||\n'
            '              part.complete)\n'
            '            return;\n'
            '          part.recoveryTimer = null;\n'
            '          _schedulePumpAll();\n'
            '        }),\n'
            '      );\n'
            '    });\n'
            '    _schedulePumpAll();\n'
            '    return true;\n'
            '  }\n'
            '\n'
            '  /// Reconcile after returning to the app, when the OS may have completed or\n'
            '  /// removed workers without delivering their final callback to Dart.\n'
            '  Future<void> reconcile(Future<Set<String>> Function() liveTaskIds) async {\n'
            '    for (final session in List<_ParallelSession>.from(_sessions.values)) {\n'
            '      await session.serialize(() async {\n'
            '        if (_disposed || !session.active || session.deleted) return;\n'
            '        final live = await liveTaskIds();\n'
            '        var recovering = false;\n'
            '        for (final part in session.parts) {\n'
            '          if (part.complete || !part.launched || part.recoveryTimer != null) {\n'
            '            continue;\n'
            '          }\n'
            '\n'
            '          final nativeOwnsPart = live.contains(part.task.taskId);\n'
            '          if (await _adoptExactSizePart(\n'
            '            session,\n'
            '            part,\n'
            '            settleNativeOwner: nativeOwnsPart,\n'
            '          )) {\n'
            '            continue;\n'
            '          }\n'
            '\n'
            '          // A genuinely live worker normally continues untouched. A live child\n'
            '          // parked at the completion sentinel is different: arm the watchdog so\n'
            '          // stale URLSession ownership cannot reserve the final slot forever.\n'
            '          if (nativeOwnsPart) {\n'
            '            _armTailStallWatch(session, part);\n'
            '            continue;\n'
            '          }\n'
            '\n'
            '          // URLSession can temporarily drop a worker during hand-off without\n'
            '          // delivering its final callback to Dart. Recover that one child;\n'
            '          // never pause healthy siblings just because ownership vanished.\n'
            '          _stabilizeSessionForRecovery(session);\n'
            '          recovering = _schedulePartRecovery(session, part) || recovering;\n'
            '        }\n'
            '        if (session.parts.every((part) => part.complete)) {\n'
            '          await _assemble(session);\n'
            '        } else {\n'
            '          if (recovering) {\n'
            '            await _status(session, TaskStatus.running);\n'
            '          }\n'
            '          await _persist(session);\n'
            '          _schedulePumpAll();\n'
            '        }\n'
            '      });\n'
            '    }\n'
            '  }\n'
            '\n'
            '  Future<bool> _pause(_ParallelSession session) async {\n'
            '    session.active = false;\n'
            '    session.generation++;\n'
            '    _speedTelemetry.resetSpeed(session.task.taskId);\n'
            '    session.cancelAggregateProgress();\n'
            '    session.cancelProgressPersist();\n'
            '    session.cancelDiskProgressPoll();\n'
            '    session.cancelCoordinatorRecovery();\n'
            '    session.resetRamp();\n'
            '\n'
            '    final unfinished = session.parts\n'
            '        .where((part) => !part.complete)\n'
            '        .toList(growable: false);\n'
            '    var pauseFailed = false;\n'
            '\n'
            '    // Pause every actual child identity concurrently. DownloadService routes\n'
            '    // these child tasks to FileDownloader.pause, which owns their URLSession\n'
            '    // resume data; it must never route them through the single-file Transfer.\n'
            '    await Future.wait(\n'
            '      unfinished.map((part) async {\n'
            '        try {\n'
            '          await pausePart(part.task);\n'
            '        } catch (_) {\n'
            '          pauseFailed = true;\n'
            '        }\n'
            '      }),\n'
            '    );\n'
            '\n'
            '    Set<String> live = <String>{};\n'
            '    final lookupLive = livePartIds;\n'
            '    if (lookupLive != null) {\n'
            '      try {\n'
            '        live = await lookupLive();\n'
            '      } catch (_) {\n'
            '        pauseFailed = true;\n'
            '      }\n'
            '    }\n'
            '\n'
            '    var stillLive = unfinished\n'
            '        .where((part) => live.contains(part.task.taskId))\n'
            '        .toList(growable: false);\n'
            '\n'
            '    // A pause acknowledgement and the URLSession state transition are\n'
            '    // asynchronous on iOS. Retry only identities that are still demonstrably\n'
            '    // live; never cancel them, because cancel can discard resume bytes.\n'
            '    if (stillLive.isNotEmpty) {\n'
            '      await Future.wait(\n'
            '        stillLive.map((part) async {\n'
            '          try {\n'
            '            await pausePart(part.task);\n'
            '          } catch (_) {\n'
            '            pauseFailed = true;\n'
            '          }\n'
            '        }),\n'
            '      );\n'
            '      await Future<void>.delayed(const Duration(milliseconds: 150));\n'
            '      if (lookupLive != null) {\n'
            '        try {\n'
            '          live = await lookupLive();\n'
            '        } catch (_) {\n'
            '          pauseFailed = true;\n'
            '        }\n'
            '      }\n'
            '      stillLive = unfinished\n'
            '          .where((part) => live.contains(part.task.taskId))\n'
            '          .toList(growable: false);\n'
            '    }\n'
            '\n'
            '    if (stillLive.isNotEmpty || (lookupLive == null && pauseFailed)) {\n'
            '      final stillIds = stillLive.map((part) => part.task.taskId).toSet();\n'
            '      session.active = true;\n'
            '      for (final part in unfinished) {\n'
            '        final owns = stillIds.contains(part.task.taskId);\n'
            '        part.launched = owns;\n'
            '        part.speed = 0;\n'
            '        if (owns) {\n'
            '          _activeConnectionIds.add(part.task.taskId);\n'
            '        } else {\n'
            '          _activeConnectionIds.remove(part.task.taskId);\n'
            '        }\n'
            '      }\n'
            '      _scheduleDiskProgressPoll(session);\n'
            '      await _persist(session);\n'
            '      await _status(session, TaskStatus.running);\n'
            '      return false;\n'
            '    }\n'
            '\n'
            '    for (final part in unfinished) {\n'
            '      part.launched = false;\n'
            '      part.speed = 0;\n'
            '    }\n'
            '    final ids = session.parts.map((part) => part.task.taskId).toSet();\n'
            '    _activeConnectionIds.removeWhere(ids.contains);\n'
            '    await _persist(session);\n'
            '    await _status(session, TaskStatus.paused);\n'
            '    _schedulePumpAll();\n'
            '    return true;\n'
            '  }\n'
            '\n'
            '  Future<void> cancel(ParallelDownloadTask task) async {\n'
            '    if (_disposed) return;\n'
            '    if (!await restore(task)) return;\n'
            '    final session = _sessions[task.taskId]!;\n'
            '    session.deleted = true;\n'
            '    session.active = false;\n'
            '    session.cancelAggregateProgress();\n'
            '    session.cancelProgressPersist();\n'
            '    session.cancelDiskProgressPoll();\n'
            '    session.cancelCoordinatorRecovery();\n'
            '    _speedTelemetry.remove(task.taskId);\n'
            '    session.resetRamp();\n'
            '    await session.serialize(() async {\n'
            '      for (final part in session.parts) {\n'
            '        _activeConnectionIds.remove(part.task.taskId);\n'
            '      }\n'
            '      await cancelParts(session.parts.map((part) => part.task.taskId).toList());\n'
            '      for (final part in session.parts) {\n'
            '        _children.remove(part.task.taskId);\n'
            '        final file = File(await part.task.filePath());\n'
            '        if (await file.exists()) await file.delete();\n'
            '      }\n'
            '      if (await session.manifest.parent.exists()) {\n'
            '        await session.manifest.parent.delete(recursive: true);\n'
            '      }\n'
            '      final staging = File(\'${await task.filePath()}.assembling\');\n'
            '      if (await staging.exists()) await staging.delete();\n'
            '      _sessions.remove(task.taskId);\n'
            '    });\n'
            '    _schedulePumpAll();\n'
            '  }\n'
            '\n'
            '  Future<void> _status(_ParallelSession session, TaskStatus status) async {\n'
            '    diagnosticLog?.record(\'parallel.status\', {\n'
            '      \'taskId\': session.task.taskId,\n'
            '      \'status\': status.name,\n'
            '      \'progress\': session.progress,\n'
            '    });\n'
            '    if (status == TaskStatus.running) {\n'
            '      session.parentRunningReported = true;\n'
            '    } else if (status == TaskStatus.enqueued || status == TaskStatus.paused) {\n'
            '      session.parentRunningReported = false;\n'
            '    }\n'
            '    await _writeParentRecord(\n'
            '      session,\n'
            '      TaskRecord(session.task, status, session.progress, session.size),\n'
            '    );\n'
            '    onUpdate(TaskStatusUpdate(session.task, status));\n'
            '  }\n'
            '\n'
            '  Future<void> _persist(_ParallelSession session) async {\n'
            '    if (session.deleted) return;\n'
            '    await session.manifest.parent.create(recursive: true);\n'
            '    final payload = jsonEncode({\n'
            '      \'schemaVersion\': kParallelManifestSchemaVersion,\n'
            '      \'generation\': session.generation,\n'
            '      \'expectedBytes\': session.size,\n'
            '      \'resourceValidator\': session.resourceValidator,\n'
            '      \'parts\': session.parts.map((part) => part.toJson()).toList(),\n'
            '    });\n'
            '    final temp = File(\'${session.manifest.path}.tmp\');\n'
            '    await temp.writeAsString(payload, flush: true);\n'
            '    try {\n'
            '      await temp.rename(session.manifest.path);\n'
            '    } on FileSystemException {\n'
            '      await session.manifest.writeAsString(payload, flush: true);\n'
            '      try {\n'
            '        if (await temp.exists()) await temp.delete();\n'
            '      } catch (_) {}\n'
            '    }\n'
            '  }\n'
            '\n'
            '  Future<bool> _adoptCompletedTarget(_ParallelSession session) async {\n'
            '    final target = File(await session.task.filePath());\n'
            '    if (await target.exists()) {\n'
            '      if (await target.length() != session.size) return false;\n'
            '      await _finishCompleteSession(session);\n'
            '      return true;\n'
            '    }\n'
            '\n'
            '    // A crash can happen after the complete staging file was flushed and\n'
            '    // closed but before its atomic rename. Reuse it only when every source\n'
            '    // Range is still exact, which proves this staging file belongs to this\n'
            '    // recoverable multipart generation. Otherwise normal assembly rewrites it.\n'
            '    final staging = File(\'${target.path}.assembling\');\n'
            '    if (!await staging.exists() || await staging.length() != session.size) {\n'
            '      return false;\n'
            '    }\n'
            '    for (final part in session.parts) {\n'
